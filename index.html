Калькулятор стоимости заказа

function addProduct() { const products = document.getElementById("products"); const row = document.createElement("div"); row.className = "product-row"; row.innerHTML = ` `; products.appendChild(row); } function removeProduct(button) { const rows = document.querySelectorAll(".product-row"); if (rows.length === 1) { document.getElementById("result").textContent = "В заказе должна остаться хотя бы одна позиция."; return; } button.closest(".product-row").remove(); document.getElementById("result").textContent = ""; } function calculateTotal() { const rows = document.querySelectorAll(".product-row"); const result = document.getElementById("result"); let total = 0; let details = ""; let hasError = false; rows.forEach(function(row) { const product = row.querySelector(".product"); const quantityInput = row.querySelector(".quantity"); const price = Number(product.value); const quantity = Number(quantityInput.value); if (!Number.isInteger(quantity) || quantity < 1) { hasError = true; quantityInput.setCustomValidity( "Введите целое количество не меньше 1." ); quantityInput.reportValidity(); return; } quantityInput.setCustomValidity(""); const cost = price * quantity; total += cost; details += `

${product.selectedOptions[0].text.split(" — ")[0]}: ${quantity} шт. × ${price} руб. = ${cost} руб.

`; }); if (hasError) { result.textContent = "Проверьте количество товаров."; return; } result.innerHTML = details + `

Итого: ${total} руб.

`; }
